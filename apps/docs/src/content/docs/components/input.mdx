title: Input

## Installation

```package-install
npx @catalix/cli@latest add input
```

## Usage

```tsx title="page.tsx" lineNumbers
import Input from "@/components/core/input"; //[!code ++]

export default function Example() {
  return <Input type="email" placeholder="Email" />; //[!code ++]
}
```
